using Lokad.Lython.Tests.Harness;

namespace Lokad.Lython.PublicApi.Tests;

public sealed class JsonStringDecodingTests
{
    // Frozen CPython 3.13.2 string-token errors, including syntax precedence
    // after unsupported surrogate values and original scalar/line positions.
    public static TheoryData<string, bool, string> Errors => new()
    {
        { "\"a\\qb\"", true, "Invalid \\escape|2|1|3|True\n" },
        { "\"\\u12\"", true, "Invalid \\uXXXX escape|2|1|3|True\n" },
        { "\"\\ud800\\uQ000\"", true, "Invalid \\uXXXX escape|8|1|9|True\n" },
        { "\"\\ud800\\q\"", true, "Invalid \\escape|7|1|8|True\n" },
        { "\"abc\\", true, "Unterminated string starting at|0|1|1|True\n" },
        { "\"a\u0000b\"", true, "Invalid control character at|2|1|3|True\n" },
        { "[\n \"café😀\\q\"]", true, "Invalid \\escape|9|2|8|True\n" },
        { "\"café😀\\u12X4\"", true, "Invalid \\uXXXX escape|7|1|8|True\n" },
        { "\"\\udc00\\q\"", true, "Invalid \\escape|7|1|8|True\n" },
        { "\"\\ud800", true, "Invalid \\uXXXX escape|2|1|3|True\n" },
        { "\"\\u0061", true, "Invalid \\uXXXX escape|2|1|3|True\n" },
        { "\"\\ud800\\udc00", true, "Invalid \\uXXXX escape|8|1|9|True\n" },
        { "\"\\ud800\\udc00\\q\"", true, "Invalid \\escape|13|1|14|True\n" },
        { "\"\\ud800\\udc00\\u000g\"", true, "Invalid \\uXXXX escape|14|1|15|True\n" },
        { "\"a\u0001b\\q\"", false, "Invalid \\escape|4|1|5|True\n" },
    };

    [Theory]
    [MemberData(nameof(Errors))]
    public async Task StringErrorsKeepOriginalDocumentCoordinates(string document, bool strict, string expected)
    {
        using var script = new LythonEngine().Compile("""
            import json
            try:
                json.loads(source, strict=strict)
            except json.JSONDecodeError as e:
                print(e.msg, e.pos, e.lineno, e.colno, e.doc is source, sep='|')
            """);
        var options = new LythonRunOptions { Globals = new Dictionary<string, object?>
            { ["source"] = document, ["strict"] = strict } };
        using var synchronous = script.Run(new MockLythonHost(), options);
        using var asynchronous = await script.RunAsync(new MockLythonHost(), options);
        foreach (var result in new[] { synchronous, asynchronous })
        {
            Assert.True(result.Success, result.Failure?.Message);
            Assert.Equal(expected, result.StandardOutput);
        }
    }

    [Theory]
    [InlineData("\\ud800")]
    [InlineData("\\udc00")]
    [InlineData("\\ud800\\u0061")]
    public async Task UnpairedSurrogateValuesFailExplicitly(string escape)
    {
        using var script = new LythonEngine().Compile("import json\nreturn json.loads(source)");
        var options = new LythonRunOptions { Globals = new Dictionary<string, object?> { ["source"] = "\"" + escape + "\"" } };
        using var synchronous = script.Run(new MockLythonHost(), options);
        using var asynchronous = await script.RunAsync(new MockLythonHost(), options);
        foreach (var result in new[] { synchronous, asynchronous })
        {
            Assert.False(result.Success);
            Assert.Equal("RuntimeError", result.Failure?.ExceptionType);
            Assert.Contains("unpaired UTF-16", result.Failure?.Message);
        }
    }
}
